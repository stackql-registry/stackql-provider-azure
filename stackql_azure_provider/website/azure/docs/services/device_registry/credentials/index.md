```sql
SELECT
id,
name,
location,
provisioningState,
systemData,
tags,
type
FROM azure.device_registry.credentials
WHERE resource_group_name = '{{ resource_group_name }}' -- required
AND namespace_name = '{{ namespace_name }}' -- required
AND subscription_id = '{{ subscription_id }}' -- required
;
```
