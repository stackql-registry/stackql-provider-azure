```sql
SELECT
name,
endTime,
error,
percentComplete,
startTime,
status
FROM azure.servicefabricmanagedclusters.operation_status
WHERE location = '{{ location }}' -- required
AND operation_id = '{{ operation_id }}' -- required
AND subscription_id = '{{ subscription_id }}' -- required
;
```
