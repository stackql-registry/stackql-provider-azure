    defaultValue="get"
    values={[

<TabItem value="get">

Get user profile.

```sql
SELECT
id,
name,
createdDate,
identity,
location,
provisioningState,
secretStore,
tags,
type,
uniqueIdentifier
FROM azure.devtestlabs.users
WHERE resource_group_name = '{{ resource_group_name }}' -- required
AND lab_name = '{{ lab_name }}' -- required
AND name = '{{ name }}' -- required
AND subscription_id = '{{ subscription_id }}' -- required
AND $expand = '{{ $expand }}'
;
```
