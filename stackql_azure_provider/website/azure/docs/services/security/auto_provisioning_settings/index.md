    defaultValue="get"
    values={[
        { label: 'get', value: 'get' },

    ]}
>
<TabItem value="get">

Details of a specific setting.

```sql
SELECT
id,
name,
autoProvision,
systemData,
type
FROM azure.security.auto_provisioning_settings
WHERE setting_name = '{{ setting_name }}' -- required
AND subscription_id = '{{ subscription_id }}' -- required
;
```
</TabItem>
